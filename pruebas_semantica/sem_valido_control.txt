class Control {
    void probar() {
        int i = 0;
        int suma = 0;
        bool activo = true;
        int opcion = 2;

        if (i < 10) {
            suma = suma + 1;
        } else if (activo) {
            suma = suma + 2;
        } else {
            suma = 0;
        }

        while (i < 10) {
            i = i + 1;
            if (i == 5) {
                continue;
            }
            if (i == 8) {
                break;
            }
        }

        do {
            i = i - 1;
            if (i < 3) {
                break;
            }
        } while (i > 0);

        for (int j = 0; j < 5; j++) {
            if (j == 3) {
                break;
            }
            suma = suma + j;
        }

        for (int j = 0; j < 2; j++) {
            suma = suma + j;
        }

        switch (opcion) {
            case 1: suma = 1; break;
            case 2: suma = 2; break;
            default: suma = 0;
        }
    }
}
