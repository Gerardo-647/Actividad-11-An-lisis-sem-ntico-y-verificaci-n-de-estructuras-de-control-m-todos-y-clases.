int cuadrado(int n) {
    return n * n;
}

float promedio(int suma, int cantidad) {
    return suma / cantidad;
}

bool esPar(int n) {
    return n % 2 == 0;
}

void saludar() {
    escribir("Hola");
    return;
}

int absoluto(int n) {
    if (n < 0) {
        return -n;
    } else {
        return n;
    }
}

int clasificar(int n) {
    if (n < 0) {
        return -1;
    } else if (n == 0) {
        return 0;
    } else {
        return 1;
    }
}

int infinito() {
    while (true) {
        return 1;
    }
}

int porOpcion(int op) {
    switch (op) {
        case 1: return 10;
        default: return 0;
    }
}

// sobrecarga: mismo nombre, distintos parametros
int sumar(int a, int b) {
    return a + b;
}

float sumar(float a, float b) {
    return a + b;
}

int sumar(int a, int b, int c) {
    return a + b + c;
}

// recursividad y uso de un metodo declarado mas abajo
int factorial(int n) {
    if (n <= 1) {
        return 1;
    }
    return n * factorial(n - 1);
}

int posterior(int x) {

}
